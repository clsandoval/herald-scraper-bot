--- SCHEMA (def init_db(conn)) ---

--- DERIVED VIEW: def match_view(raw) ---
This is the SINGLE source of truth for derived semantics. It MUST reproduce
spikes/menu-v2/render.py load_matches() per-match transform EXACTLY (the board's
load_matches() will import our reader, so parity is mandatory). `raw` is a parsed
Stratz match dict. Compute and return a dict with these keys/values:
  rk = sum(raw.get("radiantKills") or [])
  dk = sum(raw.get("direKills") or [])
  leads = raw.get("radiantNetworthLeads") or [0]
  players (list, one per raw player, in raw order): each dict =
    {"hero_id": p["heroId"], "is_radiant": p["isRadiant"], "k": p["kills"],
     "d": p["deaths"], "a": p["assists"], "gpm": p.get("goldPerMinute", 0),
     "networth": p.get("networth", 0),
     "items": [i for i in [p.get(f"item{i}Id") for i in range(6)] if i],
     "lane": p.get("lane"), "role": p.get("role"), "position": p.get("position"),
     "networth_per_min": (p.get("stats") or {}).get("networthPerMinute") or [],
     "purchases": (p.get("stats") or {}).get("itemPurchases") or []}
  feeder = max(players, key=lambda p: p["d"])
  win_r = raw["didRadiantWin"]
  max_lead, min_lead = max(leads), min(leads)
  deficit = -min_lead if win_r else max_lead   # winner's worst deficit
  return {"id": raw["id"], "duration": raw["durationSeconds"],
    "mins": raw["durationSeconds"] // 60, "start": raw.get("startDateTime"),
    "radiant_win": win_r, "kills_r": rk, "kills_d": dk, "kills": rk + dk,
    "kpm": round((rk + dk) / max(raw["durationSeconds"] / 60, 1), 1),
    "bracket": raw.get("bracket"), "rank": raw.get("rank"),
    "leads": leads, "max_lead": max_lead, "min_lead": min_lead,
    "comeback_gold": deficit if deficit > 0 else 0,
    "feeder": feeder, "players": players,
    "tower_deaths": raw.get("towerDeaths") or [],
    "lanes": {"top": raw.get("topLaneOutcome"), "mid": raw.get("midLaneOutcome"),
              "bot": raw.get("bottomLaneOutcome")}}

--- COLUMN COMPUTE: def derived_cols(raw) ---
Returns the extra scalar columns for the matches row (used by SQL sort/filter). All
formulas per locked decisions; reuse match_view where handy:
  v = match_view(raw)
  throw_size = v["max_lead"] if not v["radiant_win"] else -v["min_lead"]  # loser's best lead
  throw_gold = max(0, throw_size)
  feeder_deaths = max((p["d"] for p in v["players"]), default=0)
  top_kills = max((p["k"] for p in v["players"]), default=0)
  max_gpm = max((p["gpm"] for p in v["players"]), default=0)
  winner_towers_lost = sum(1 for t in v["tower_deaths"] if t.get("isRadiant") == v["radiant_win"])
  has_rapier = 1 if any(i == RAPIER_ID for p in v["players"] for i in p["items"]) else 0
  final_lead = v["leads"][-1]
  return all of: kills_r, kills_d, kills, kpm, max_lead, min_lead, final_lead,
    comeback_gold(=v["comeback_gold"]), throw_gold, feeder_deaths, top_kills,
    max_gpm, winner_towers_lost, has_rapier, final_lead.

--- UPSERT: def upsert_match(conn, raw, avg_rank_tier) ---
Shared by live enrichment AND selfcheck. mid = raw["id"]. c = derived_cols(raw).
INSERT OR REPLACE INTO matches (all 24 columns) VALUES(...): match_id=mid,
start_time=raw.get("startDateTime"), duration_s=raw["durationSeconds"],
radiant_win=1 if raw["didRadiantWin"] else 0, game_mode=raw.get("gameMode"),
rank=raw.get("rank"), bracket=raw.get("bracket"), avg_rank_tier=avg_rank_tier,
kills_r..has_rapier from c, raw=json.dumps(raw), enriched_at=int(time.time()).
Then for players: DELETE FROM match_players WHERE match_id=mid, then for slot,p in
enumerate(raw["players"]): INSERT OR REPLACE INTO match_players VALUES(mid, slot,
p["heroId"], 1 if p["isRadiant"] else 0, p["kills"], p["deaths"], p["assists"],
p.get("networth",0), p.get("goldPerMinute",0), p.get("position"),
json.dumps([i for i in [p.get(f"item{i}Id") for i in range(6)] if i])).
conn.commit().

--- READER: def load_matches(conn, where="", params=()) ---
The board imports THIS. Returns list of match_view dicts, newest first:
  sql = "SELECT raw FROM matches" + (f" WHERE {where}" if where else "") + " ORDER BY start_time DESC"
  rows = conn.execute(sql, params).fetchall()
  return [match_view(json.loads(r[0])) for r in rows]

--- OD HTTP: def od_fetch(client, less_than) ---
GET OD_URL with params {"less_than_match_id": less_than} if less_than else {}.
Retry-with-backoff on 5xx/timeout (OD origin flaps — FINDINGS): up to 5 attempts,
on 429 sleep 10s and retry, on 5xx/exception log warning + sleep 3s and retry,
raise_for_status on other codes. Return parsed JSON list, or [] if all attempts fail.

--- DISCOVERY: def discover(conn) ---
  pre = set of match_id from `SELECT match_id FROM matches` UNION `SELECT match_id FROM pending`
  seen_new = set(); less_than = None; new = 0; now = int(time.time())
  with httpx.Client(timeout=30) as client:
    for page_i in range(MAX_PAGES):
      rows = od_fetch(client, less_than)
      if not rows: break
      page_ids = [m["match_id"] for m in rows]
      novel = [m for m in rows if m["match_id"] not in pre and m["match_id"] not in seen_new]
      for m in novel:
        seen_new.add(m["match_id"])
        rt = m.get("avg_rank_tier")
        if rt is not None and 10 <= rt <= 15:
          conn.execute("INSERT OR IGNORE INTO pending(match_id,avg_rank_tier,start_time,discovered_at,attempts,last_attempt) VALUES(?,?,?,?,0,NULL)",
                       (m["match_id"], rt, m.get("start_time"), now)); new += 1
      conn.commit()
      if not novel:  # full page yielded only already-known ids -> reached frontier
        break
      less_than = min(page_ids)
      time.sleep(1.1)
  return new

--- STRATZ HTTP: def stratz_fetch(client, match_id) ---
Read token lazily: tok = os.environ.get("STRATZ_API_TOKEN"); if not tok: raise
RuntimeError("STRATZ_API_TOKEN not set"). headers = {"Authorization": f"Bearer {tok.strip()}",
"User-Agent": "STRATZ_API"} — BOTH required (Stratz 403s without UA). body =
{"query": STRATZ_QUERY, "variables": {"id": match_id}}. POST STRATZ_URL. On 429:
sleep 30s and retry (per locked decision). Up to 3 attempts; on exception log +
sleep 3s + retry. On success parse json; if data.get("errors") log warning;
return (data.get("data") or {}).get("match"). NEVER log the token or headers.

--- ENRICHMENT: def enrich(conn) ---
  now = int(time.time()); enriched = 0; failed = 0; dropped = 0
  rows = conn.execute("SELECT match_id, avg_rank_tier FROM pending WHERE discovered_at < ? AND attempts < 8 ORDER BY discovered_at ASC LIMIT 120", (now - 180,)).fetchall()
  with httpx.Client(timeout=30) as client:
    for mid, art in rows:
      m = stratz_fetch(client, mid)
      ready = bool(m) and bool(m.get("radiantNetworthLeads"))
      if ready:
        upsert_match(conn, m, art)
        conn.execute("DELETE FROM pending WHERE match_id=?", (mid,)); enriched += 1
      else:
        conn.execute("UPDATE pending SET attempts=attempts+1, last_attempt=? WHERE match_id=?", (now, mid))
        att = conn.execute("SELECT attempts FROM pending WHERE match_id=?", (mid,)).fetchone()[0]
        if att >= 8:
          conn.execute("DELETE FROM pending WHERE match_id=?", (mid,)); dropped += 1
          log.info(f"dropped {mid} after {att} attempts")
        failed += 1
      conn.commit()
      time.sleep(0.6)  # under Stratz 150/min
  return enriched, failed, dropped

--- PRUNE: def prune(conn) ---
  now = int(time.time()); cutoff = now - RETENTION_DAYS * 86400
  old = [r[0] for r in conn.execute("SELECT match_id FROM matches WHERE start_time < ?", (cutoff,)).fetchall()]
  conn.executemany("DELETE FROM match_players WHERE match_id=?", [(x,) for x in old])
  conn.execute("DELETE FROM matches WHERE start_time < ?", (cutoff,))
  conn.execute("DELETE FROM pending WHERE discovered_at < ?", (now - 2 * 86400,))
  conn.commit(); return len(old)

--- CYCLE: def cycle(conn) ---
  n_disc = discover(conn); n_enr, n_fail, n_drop = enrich(conn); n_prune = prune(conn)
  mtot = conn.execute("SELECT COUNT(*) FROM matches").fetchone()[0]
  ptot = conn.execute("SELECT COUNT(*) FROM pending").fetchone()[0]
  log.info(f"cycle: discovered {n_disc}, enriched {n_enr}, not-ready {n_fail}, dropped {n_drop}, pruned {n_prune} | matches={mtot} pending={ptot}")
